# StarHermit engine script game: one package, played natively by the
# StarHermit client and in browsers by the engine's web runtime. Browsers open
# launch=index.html, the loader `npm run package` writes (it pins the engine
# on wiki.starhermit.com by hash; never edit it). Re-run `npm run package`
# after every change and commit index.html + starhermit-files.json.
launch=index.html
engine.api=1
engine.entry=native/main.ts
engine.version=1.30.0
engine.minor=32
engine.capabilities=platform,storage,clipboard

# Keyboard controls (players rebind keys and pads in the runtime's Options).
# The board is also played with the pointer or a finger: tap a piece in the
# tray (or the 3D offer row), then tap a cell — or drag a piece onto the board.
engine.group.cursor_up=Board cursor
control.cursor_up=ArrowUp | Cursor up
control.cursor_down=ArrowDown | Cursor down
control.cursor_left=ArrowLeft | Cursor left
control.cursor_right=ArrowRight | Cursor right
engine.group.slot_1=Pieces
control.slot_1=Digit1+Numpad1 | Select piece 1
control.slot_2=Digit2+Numpad2 | Select piece 2
control.slot_3=Digit3+Numpad3 | Select piece 3
control.next_slot=KeyN | Next piece
control.place=Enter+NumpadEnter+Space | Place piece
control.clear=KeyC | Clear selection
engine.group.hint=Round
control.hint=KeyH | Hint
control.undo=KeyU | Undo
control.pause=Escape | Pause / close

# Gamepad defaults (as the browser build's gamepad poll): the D-pad or left
# stick moves the cursor, A places, B clears the selection, X cycles the
# offered pieces, Start pauses; added: Y hints, the left shoulder undoes.
engine.gamepad.cursor_up=DpadUp+LeftStickUp
engine.gamepad.cursor_down=DpadDown+LeftStickDown
engine.gamepad.cursor_left=DpadLeft+LeftStickLeft
engine.gamepad.cursor_right=DpadRight+LeftStickRight
engine.gamepad.place=South
engine.gamepad.clear=East
engine.gamepad.next_slot=West
engine.gamepad.hint=North
engine.gamepad.undo=LeftShoulder
engine.gamepad.pause=Start
